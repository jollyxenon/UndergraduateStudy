\titledquestion{Compare and Prove}

For each pair, determine which of \(f(n)=O(g(n))\), \(f(n)=\Omega(g(n))\), and \(f(n)=\Theta(g(n))\) hold. Prove your answer using the definitions or limits. If the growth rates differ strictly, also state the corresponding little-\(o\) or little-\(\omega\) relation. All logarithms have base 2, and \(n\) tends to infinity through positive integers.

\begin{parts}
\part[3] \(f(n)=1.01^n\) and \(g(n)=n^{10}\).
\begin{solution}

$$
L = \lim_{n\to\infty} \frac{f(n)}{g(n)}
= \lim_{n\to\infty} \frac{1.01^n}{n^{10}}
= \infty
$$

$$
f(n) = \omega(g(n))
$$

And $f(n) = \Omega(g(n))$, but $f(n) \neq O(g(n))$, $f(n) \neq \Omega(g(n))$. 

\end{solution}

\part[3] \(f(n)=\log(n!)\) and \(g(n)=\log(n^n)\).
\begin{solution}

$$
g(n) = \log(n^n) = n\log n
$$

$$
n! < n^n \implies \log(n!) < \log(n^n) = n\log n
$$

When $n$ is large enough,

$$
\log(n!)
= \sum_{i=1}^{n} \log i
> \sum_{i=n/2}^{n} \log i
> \frac{n}{2} \log(n/2)
= \frac{n}{2}(\log n - 1)
$$

According to the general squeeze theorem, we have

$$
f(n) = \Theta(g(n))
$$

So $f(n) = \Omega(g(n))$ and $f(n) = O(g(n))$ also hold.

What's more, $f(n) \neq o(g(n))$ and $f(n) \neq \omega(g(n))$.

\end{solution}

\part[4] \(f(n)=n^{1+\varepsilon}\) and \(g(n)=n(\log n)^k\), where \(\varepsilon>0\) and the positive integer \(k\) are fixed constants.
\begin{solution}

$$
L
= \lim_{n\to\infty} \frac{f(n)}{g(n)}
= \lim_{n\to\infty} \frac{n^{1+\varepsilon}}{n(\log n)^k}
= \lim_{n\to\infty} \frac{n^{\varepsilon}}{(\log n)^k}
$$

Let $n=2^t$, then

$$
\frac{n^{\varepsilon}}{(\log_2 n)^k}
= \frac{(2^t)^\varepsilon}{t^k}
= \frac{(2^\varepsilon)^t}{t^k}
$$

$\varepsilon > 0$, so $2^\varepsilon > 1$, and thus $\frac{(2^\varepsilon)^t}{t^k} \to \infty$ as $t \to \infty$.

$$
L = \infty
$$

$$
f(n) = \omega(g(n))
$$

So $f(n) = \Omega(g(n))$,but $f(n) \neq O(g(n))$, $f(n) \neq \Theta(g(n))$.

What's more, $f(n) \neq o(g(n))$.

\end{solution}
\end{parts}
