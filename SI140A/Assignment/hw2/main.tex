\begin{homeworkProblem}

A box contains 5 red balls, 4 blue balls, and 3 green balls. Three balls are drawn sequentially without replacement.

Let
$
A=\{\text{the first ball is red}\},
$
and let
$
C=\{\text{at least one of the three drawn balls is red}\}.
$

Suppose that the only information revealed after the experiment is that event $C$ occurred; that is, at least one of the three drawn balls was red.

\begin{enumerate}
    \item[(a)] Find $P(C)$.

    \item[(b)] Find $P(A\cap C)$ and hence calculate $P(A\mid C)$.

    \item[(c)] Let $B$ be the event that the second ball is red. Calculate $P(B\mid C)$.

    \item[(d)] Calculate $P(A\cap B\mid C)$.

    \item[(e)] Compare $P(A\mid C)$ with $P(A)$. Are $A$ and $C$ independent? Justify your answer using the definition of independence.
\end{enumerate}

\textbf{Solution: }

\begin{enumerate}
    \item[(a)]
    
    $$
    P(C)
    = 1-P(\neg C)
    = 1-P(\text{no red balls are drawn})
    = 1-\frac{\binom{7}{3}}{\binom{12}{3}}
    = \frac{37}{44}
    $$

    \item[(b)]
    
    $$
    P(A \cap C)
    = P(A)
    = \frac{5}{12}
    $$

    $$
    P(A \mid C)
    = \frac{P(A \cap C)}{P(C)}
    = \frac{\frac{5}{12}}{\frac{37}{44}}
    = \frac{55}{111}
    $$

    \item[(c)]
    
    $$
    P(B \cap C)
    = P(B)
    = \frac{5}{12}
    $$

    $$
    P(B \mid C)
    = \frac{P(B \cap C)}{P(C)}
    = \frac{\frac{5}{12}}{\frac{37}{44}}
    = \frac{55}{111}
    $$

    \item[(d)]
    
    $$
    P(A \cap B \cap C)
    = P(A \cap B)
    = \frac{5}{12} \cdot \frac{4}{11}
    = \frac{5}{33}
    $$

    $$
    P(A \cap B \mid C)
    = \frac{P(A \cap B \cap C)}{P(C)}
    = \frac{\frac{5}{33}}{\frac{37}{44}}
    = \frac{20}{111}
    $$

    \item[(e)]
    
    $$
    P(A \mid C)
    = \frac{55}{111}
    \neq P(A)
    = \frac{5}{12}
    $$

    Therefore, $A$ and $C$ are not independent.

    If $A$ and $C$ were independent, then $P(A \cap C) = P(A) P(C)$.

    $$
    P(A \cap C) = P(A \mid C) P(C)
    $$

    So if $A$ and $C$ were independent, then $P(A \mid C) = P(A)$.

    What's more, intuitively, when $C$ is known, the occasion of not drawing a red ball is eliminated, which increases the probability of drawing a red ball in the first draw.
\end{enumerate}

\end{homeworkProblem}

\begin{homeworkProblem}
\textbf{Multiple Detection System}

A monitoring system uses four independent sensors to detect whether a machine has experienced a failure. Let $A_i$ denote the event that sensor $i$ raises a false alarm.

The false-alarm probabilities are
$
P(A_1)=0.10,\;
P(A_2)=0.15,\;
P(A_3)=0.20,\;
P(A_4)=0.25.
$

Assume that the four events are mutually independent.

The monitoring system issues a warning if \textbf{at least one sensor} raises a false alarm.

\begin{enumerate}[(a)]
    \item Suppose the monitoring system has issued a warning. Find the probability that \textbf{at least two sensors} have raised a false alarm, i.e.,
    $
    P\left(
    \text{at least two alarms}
    \,\middle|\,
    A_1\cup A_2\cup A_3\cup A_4
    \right).
    $

    \item Suppose now that no independence assumption is made. Instead, you are given
    $
    P(A_i)=0.1,\; i=1,2,3,4,
    $
    and
    $
    P(A_i\cap A_j)=0.02
    $
    for every $i\neq j$, together with
    $
    P(A_i\cap A_j\cap A_k)=0.005
    $
    for every distinct $i,j,k$, while
    $
    P(A_1\cap A_2\cap A_3\cap A_4)=0.001.
    $

    Use the inclusion-exclusion principle to calculate
    $
    P(A_1\cup A_2\cup A_3\cup A_4).
    $
\end{enumerate}

\textbf{Solution: }

\begin{enumerate}
    \item[(a)]
    
    $$
    \begin{aligned}
    &P(A_1 \cup A_2 \cup A_3 \cup A_4) \\
    =& 1 - P(\neg A_1 \cap \neg A_2 \cap \neg A_3 \cap \neg A_4) \\
    =& 1 - P(\neg A_1) P(\neg A_2) P(\neg A_3) P(\neg A_4) \\
    =& 1 - (1-0.1)(1-0.15)(1-0.2)(1-0.25) \\
    =& 0.541
    \end{aligned}
    $$

    $$
    \begin{aligned}
    &P(\text{at least two alarms}) \\
    =& 1 - P(\text{no alarm}) - P(\text{exactly one alarm}) \\
    =& 1 - P(\neg A_1 \cap \neg A_2 \cap \neg A_3 \cap \neg A_4) - \sum_{i=1}^{4} P(A_i \cap \bigcap_{j\neq i} \neg A_j) \\
    =& 1 - (1-0.1)(1-0.15)(1-0.2)(1-0.25) - \sum_{i=1}^{4} P(A_i) \prod_{j\neq i} P(\neg A_j) \\
    =& 0.14125
    \end{aligned}
    $$

    $$
    \begin{aligned}
    &P(\text{at least two alarms} \mid A_1 \cup A_2 \cup A_3 \cup A_4) \\
    =& \frac{P(\text{at least two alarms} \cap (A_1 \cup A_2 \cup A_3 \cup A_4))}{P(A_1 \cup A_2 \cup A_3 \cup A_4)} \\
    =& \frac{P(\text{at least two alarms})}{P(A_1 \cup A_2 \cup A_3 \cup A_4)} \\
    =& \frac{0.14125}{0.541} \\
    =& \frac{565}{2164} \\
    \approx& 0.2611
    \end{aligned}
    $$

    \item[(b)]
    
    $$
    P\left(\bigcup_{i=1}^4 A_i\right)
    =
    \sum_{1 \le i \le 4} P(A_i) - \sum_{1 \le i < j \le 4} P(A_i \cap A_j) + \sum_{1 \le i < j < k \le 4} P(A_i \cap A_j \cap A_k) - P(A_1 \cap A_2 \cap A_3 \cap A_4)
    $$

    $$
    \sum P(A_i) = 4 \times 0.1 = 0.4
    $$

    $$
    \sum P(A_i \cap A_j) = 6 \times 0.02 = 0.12
    $$

    $$
    \sum P(A_i \cap A_j \cap A_k) = 4 \times 0.005 = 0.020
    $$

    $$
    P(A_1 \cap A_2 \cap A_3 \cap A_4) = 0.001
    $$

    $$
    P(A_1 \cup A_2 \cup A_3 \cup A_4) = 0.4 - 0.12 + 0.020 - 0.001 = 0.299
    $$
\end{enumerate}
\end{homeworkProblem}

\begin{homeworkProblem}

A factory has three production lines, $L_1$, $L_2$, and $L_3$, which produce $30\%$, $45\%$, and $25\%$ of the factory's total products, respectively. The probabilities that a product is defective given that it comes from $L_1$, $L_2$, or $L_3$ are
$
P(D\mid L_1)=0.02,\;
P(D\mid L_2)=0.03,\;
P(D\mid L_3)=0.05,
$
where $D$ denotes the event that a randomly selected product is defective.

\begin{enumerate}

    \item[(a)] Two products are selected independently from the factory. Find the probability that both products are defective.

    \item[(b)] Three products are selected independently from the factory. Find the probability that at least one of the three products is defective.

    \item[(c)] Suppose $n$ products are selected independently from the factory. Find the smallest integer $n$ such that the probability of obtaining at least one defective product is at least $95\%$.

\end{enumerate}

\textbf{Solution: }

\begin{enumerate}
    \item[(a)]
    
    $$
    P(L_1) = 0.3,\quad P(L_2) = 0.45,\quad P(L_3) = 0.25
    $$

    $$
    P(D)
    = P(D\mid L_1)P(L_1) + P(D\mid L_2)P(L_2) + P(D\mid L_3)P(L_3)
    = 0.032
    $$

    $$
    P(\text{both defective})
    = P(D)^2
    = 0.032^2
    = 0.001024
    $$

    \item[(b)]
    
    $$
    P(\text{at least one defective})
    = 1 - P(\text{none defective})
    = 1 - (1-P(D))^3
    = 1 - (1-0.032)^3
    = 0.092960768
    $$

    \item[(c)]
    
    $$
    P(\text{at least one defective})
    = 1 - (1-P(D))^n
    = 1 - 0.968^n
    \ge 0.95
    $$

    $$
    n \ge \frac{\log(0.05)}{\log(0.968)}
    \approx 92.11
    $$

    Therefore, the smallest $n$ is $93$.
\end{enumerate}

\end{homeworkProblem}

\begin{homeworkProblem}

Two dice are rolled independently. For each die, the probability of
showing an even number is $p$, where $0<p<1$. The two dice have the
same distribution, but no assumption is made that the individual
outcomes $1,2,\ldots,6$ are equally likely.

Define
$
A=\{\text{the sum of the two dice is even}\},
$
and
$
B=\{\text{the sum of the two dice is greater than }8\}.
$

Determine whether there exists a value of $p$ such that, for every
possible distribution of the individual die outcomes satisfying
$P(\text{even})=p$, the events $A$ and $B$ are independent.

If such a value of $p$ exists, find all such values and prove that the
independence holds for every admissible distribution.

If no such value exists, prove this by constructing, for each candidate
value of $p$, two admissible distributions that have the same value of
$p$ but give different values of
$
P(A\cap B)-P(A)P(B).
$

Your argument should distinguish carefully between independence that is
implied by the given information and independence that occurs only for
a particular choice of the underlying distribution.

\textbf{Solution: }

$$
P(A)
= P(\text{both even}) + P(\text{both odd})
= p^2 + (1-p)^2
= 2p^2 - 2p + 1
$$

$$
P(A \cap B)
= P(sum = 10) + P(sum = 12)
= 2p_4 p_6 + p_5^2 + p_6^2
$$

Construct two distributions:

\begin{enumerate}
    \item[1. ]
    
    Let $p_1 = 1-p$, $p_2 = p$, $p_3 = p_4 = p_5 = p_6 = 0$.

    $$
    P(B) = 0, \quad P(A \cap B) = 0
    $$

    Therefore, 
    
    $$
    \Delta = P(A \cap B) - P(A)P(B) = 0 - P(A) \cdot 0 = 0
    $$

    \item[2. ]
    
    Let $p_1 = 1 - p$, $p_6 = p$, $p_2 = p_3 = p_4 = p_5 = 0$.

    $$
    P(B) = p^2,\quad P(A \cap B) = p^2,\quad P(A) = 2p^2 - 2p + 1
    $$

    $$
    \Delta
    = P(A \cap B) - P(A)P(B)
    = 2 p^3 (1 - p)
    > 0
    $$
\end{enumerate}

As a result, for any $0<p<1$, we can always find two distributions that give different values of $\Delta$.

Therefore, there is no value of $p$ such that $A$ and $B$ are independent for every admissible distribution.

\end{homeworkProblem}
